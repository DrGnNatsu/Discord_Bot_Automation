### 💻 Solution Snippet: The Session Manager

Here is the code for `src/bot/session_manager.py` to get you started.

```python
from src.database.db import SessionLocal
from src.database.models import ActiveSession

def get_user_session(user_id):
    """Checks if the user is currently inside a workflow."""
    session = SessionLocal()
    try:
        return session.query(ActiveSession).filter(ActiveSession.user_id == str(user_id)).first()
    finally:
        session.close()

def create_or_update_session(user_id, workflow_id, next_state):
    """Moves the user to the next state."""
    session = SessionLocal()
    try:
        # Check if exists
        active = session.query(ActiveSession).filter(ActiveSession.user_id == str(user_id)).first()
        
        if active:
            active.current_state = next_state
            active.workflow_id = workflow_id # Keep tracking source flow
        else:
            new_session = ActiveSession(
                user_id=str(user_id), 
                workflow_id=workflow_id, 
                current_state=next_state
            )
            session.add(new_session)
        
        session.commit()
        print(f"💾 Session Updated: User {user_id} -> {next_state}")
    except Exception as e:
        print(f"🔥 DB Error: {e}")
    finally:
        session.close()

def delete_session(user_id):
    """Frees the user."""
    session = SessionLocal()
    try:
        session.query(ActiveSession).filter(ActiveSession.user_id == str(user_id)).delete()
        session.commit()
    finally:
        session.close()

```
