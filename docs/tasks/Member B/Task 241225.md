## 💻 The Solution Code (`src/bot/actions.py`)

Here is the complete implementation. It includes a helper function to parse time durations (essential for the `TIMEOUT_USER` rule).

```python
import discord
from datetime import timedelta
import re

# --- HELPER: Time Parser ---
def parse_duration(duration_str: str):
    """
    Converts strings like '1h', '30m', '1d' into timedelta objects.
    """
    regex = re.match(r"(\d+)([smhd])", duration_str)
    if not regex:
        return timedelta(minutes=5) # Default fallback

    value, unit = regex.groups()
    value = int(value)
    
    if unit == 's': return timedelta(seconds=value)
    elif unit == 'm': return timedelta(minutes=value)
    elif unit == 'h': return timedelta(hours=value)
    elif unit == 'd': return timedelta(days=value)
    return timedelta(minutes=5)


# --- ACTION FUNCTIONS ---

async def send_message(event: discord.Message, params: dict):
    """
    Executes: ACTION: SEND_MESSAGE channel=general content="Hello"
    """
    guild = event.guild
    target_channel_id = params.get("channel")
    content = params.get("content")

    if not target_channel_id or not content:
        return

    # Try to find channel by ID or Name
    channel = discord.utils.get(guild.text_channels, name=target_channel_id)
    if not channel:
        # If ID was passed as string, try converting to int
        try:
            channel = guild.get_channel(int(target_channel_id))
        except ValueError:
            pass
            
    if channel:
        await channel.send(content)


async def reply_message(event: discord.Message, params: dict):
    """
    Executes: ACTION: REPLY_MESSAGE content="Hello"
    """
    content = params.get("content")
    if content:
        await event.channel.send(content, reference=event)


async def ban_user(event: discord.Message, params: dict):
    """
    Executes: ACTION: BAN_USER
    """
    # Safety: Don't ban the bot owner or admins (Basic protection)
    if event.author.guild_permissions.administrator:
        return 

    try:
        await event.author.ban(reason="GuildFlow Automated Ban")
    except discord.Forbidden:
        print(f"Error: Bot lacks permission to ban {event.author}")


async def timeout_user(event: discord.Message, params: dict):
    """
    Executes: ACTION: TIMEOUT_USER duration="1h"
    """
    duration_str = params.get("duration", "5m")
    duration_obj = parse_duration(duration_str)

    try:
        await event.author.timeout(duration_obj, reason="GuildFlow Automated Timeout")
    except discord.Forbidden:
        print(f"Error: Bot lacks permission to timeout {event.author}")


# --- THE COMMAND MAP ---
# This is what the Bot Engine will import tomorrow
COMMAND_MAP = {
    "SEND_MESSAGE": send_message,
    "REPLY_MESSAGE": reply_message,
    "BAN_USER": ban_user,
    "TIMEOUT_USER": timeout_user
}

```
