```python
import discord
from src.database.db import SessionLocal
from src.database.models import Workflow
from src.bot.actions import COMMAND_MAP

# 1. LOGIC EVALUATOR (Simple MVP Version)
def evaluate_condition(condition, event):
    """
    Evaluates: user.strikes >= 3
    For MVP, we support basic text/author checks.
    """
    if not condition: return True
    
    # Extract values
    left_str = condition["left"]   # e.g. "message.content"
    op = condition["operator"]     # e.g. "=="
    right_val = condition["right"] # e.g. "!help"

    # Resolve "left" variable
    actual_value = None
    if left_str == "message.content":
        actual_value = event.content
    elif left_str == "author.id":
        actual_value = str(event.author.id)
    # (Add more variables here as needed)

    # Compare
    if op == "==": return actual_value == right_val
    if op == "!=": return actual_value != right_val
    if op == "contains": return right_val in (actual_value or "")
    
    return False

# 2. THE DISPATCHER (Recursive)
async def execute_instructions(actions_list, event):
    """
    The Core Engine: Iterates through JSON and runs Python code.
    """
    for step in actions_list:
        step_type = step.get("type")

        # CASE A: Standard Action (SEND, BAN, etc.)
        if step_type == "ACTION":
            cmd_name = step.get("command")
            params = step.get("params", {})
            
            # Look up the function in the Command Map
            func = COMMAND_MAP.get(cmd_name)
            if func:
                print(f"⚙️ Executing {cmd_name}...")
                await func(event, params)
            else:
                print(f"⚠️ Unknown command: {cmd_name}")

        # CASE B: Logic Branch (IF/ELSE)
        elif step_type == "LOGIC_IF":
            condition = step.get("condition")
            # Evaluate the condition
            if evaluate_condition(condition, event):
                # RECURSION: Run the 'body' actions
                await execute_instructions(step.get("body", []), event)
            else:
                # Run 'else' actions if they exist (optional)
                else_body = step.get("else_body", [])
                if else_body:
                    await execute_instructions(else_body, event)

# 3. THE BOT CLIENT
class GuildFlowBot(discord.Bot):
    async def on_ready(self):
        print(f"✅ Bot Online: {self.user.name}")

    async def on_message(self, message):
        # Prevent infinite loops
        if message.author.bot: return

        # A. Database Query
        session = SessionLocal()
        try:
            # Find all workflows triggered by 'message'
            workflows = session.query(Workflow).filter(
                Workflow.trigger_event == "message",
                Workflow.is_active == True
            ).all()

            # B. Processing
            for flow in workflows:
                data = flow.compiled_json
                
                # Check Global Filter (WHERE clause) if exists
                if data.get("filter"):
                    if not evaluate_condition(data["filter"], message):
                        continue

                # C. Execution
                print(f"🚀 Triggering Workflow: {flow.name}")
                await execute_instructions(data["actions"], message)

        except Exception as e:
            print(f"🔥 Runtime Error: {e}")
        finally:
            session.close()

# Initialize variable for main.py to import
bot = GuildFlowBot(intents=discord.Intents.all())

```
