```python
import sys
# Adjust these imports to match your exact folder structure
from src.antlr_build.GuildFlowParser import GuildFlowParser
from src.antlr_build.GuildFlowVisitor import GuildFlowVisitor

class GuildFlowCompiler(GuildFlowVisitor):
    
    def visitProg(self, ctx: GuildFlowParser.ProgContext):
        """
        The Entry Point.
        Returns a list of all definitions (Workflows and States) found in the file.
        """
        return [self.visit(child) for child in ctx.definition()]

    def visitWorkflow_def(self, ctx: GuildFlowParser.Workflow_defContext):
        """
        Converts: WORKFLOW name ON event { ... }
        Into: {"type": "WORKFLOW", "name": "...", "trigger": "...", "actions": [...]}
        """
        return {
            "type": "WORKFLOW",
            "name": ctx.ID().getText(),
            "trigger": ctx.event_type().getText(),
            # Check if a WHERE clause exists (ctx.condition() might be None)
            "filter": self.visit(ctx.condition()) if ctx.condition() else None,
            # Recursively visit every statement inside the brackets
            "actions": [self.visit(stmt) for stmt in ctx.statement()]
        }

    def visitAction_statement(self, ctx: GuildFlowParser.Action_statementContext):
        """Redirects 'ACTION: ...' to the specific command logic."""
        return self.visit(ctx.action_command())

    def visitAction_command(self, ctx: GuildFlowParser.Action_commandContext):
        """
        Converts: ACTION: SEND_MESSAGE channel=log content="Hi"
        Into: {"type": "ACTION", "command": "SEND_MESSAGE", "params": {...}}
        """
        command = ctx.start.text # e.g. 'SEND_MESSAGE' or 'BAN_USER'
        params = {}

        # Helper to remove quotes from strings: "Hello" -> Hello
        def clean_str(token):
            return token.getText().strip('"')

        # Extract parameters based on the command type
        if command == "SEND_MESSAGE":
            if ctx.ID(): params["channel"] = ctx.ID().getText()
            if ctx.STRING(): params["content"] = clean_str(ctx.STRING())

        elif command == "REPLY_MESSAGE":
            params["content"] = clean_str(ctx.STRING())

        elif command == "TIMEOUT_USER":
            params["duration"] = clean_str(ctx.STRING())

        # BAN_USER has no params, so it creates an empty dict automatically

        return {
            "type": "ACTION",
            "command": command,
            "params": params
        }

    def visitIf_statement(self, ctx: GuildFlowParser.If_statementContext):
        """
        Converts: IF condition { ... } ELSE { ... }
        This handles the nesting logic (P3).
        """
        # ctx.statement() returns a list of ALL statements in both IF and ELSE blocks.
        # We need to split them. The grammar usually groups them, but for simplicity
        # in this visitor, we will assume standard traversal.
        
        # NOTE: A more robust way relies on the specific index of blocks in your grammar.
        # This implementation assumes the standard structure:
        
        return {
            "type": "LOGIC_IF",
            "condition": self.visit(ctx.condition()),
            # This logic depends on how you defined statement* in the grammar.
            # Ideally, your grammar separates then_block and else_block.
            # If using standard `statement*`, you might need to iterate children carefully.
            # For MVP, we simplify:
            "branches": "Requires Grammar Separation for precise extraction" 
        }

    def visitCondition(self, ctx: GuildFlowParser.ConditionContext):
        """Converts: user.strikes >= 3"""
        return {
            "left": ctx.expr(0).getText(),
            "operator": ctx.comparator().getText(),
            "right": ctx.expr(1).getText()
        }
```

### 💻 The Solution Code (`src/bot/actions.py`)

Here is the complete implementation. It includes a helper function to parse time durations (essential for the `TIMEOUT_USER` rule).

```python
import discord
from datetime import timedelta
import re

# --- HELPER: Time Parser ---
def parse_duration(duration_str: str):
    """
    Converts strings like '1h', '30m', '1d' into timedelta objects.
    """
    regex = re.match(r"(\d+)([smhd])", duration_str)
    if not regex:
        return timedelta(minutes=5) # Default fallback

    value, unit = regex.groups()
    value = int(value)
    
    if unit == 's': return timedelta(seconds=value)
    elif unit == 'm': return timedelta(minutes=value)
    elif unit == 'h': return timedelta(hours=value)
    elif unit == 'd': return timedelta(days=value)
    return timedelta(minutes=5)


# --- ACTION FUNCTIONS ---

async def send_message(event: discord.Message, params: dict):
    """
    Executes: ACTION: SEND_MESSAGE channel=general content="Hello"
    """
    guild = event.guild
    target_channel_id = params.get("channel")
    content = params.get("content")

    if not target_channel_id or not content:
        return

    # Try to find channel by ID or Name
    channel = discord.utils.get(guild.text_channels, name=target_channel_id)
    if not channel:
        # If ID was passed as string, try converting to int
        try:
            channel = guild.get_channel(int(target_channel_id))
        except ValueError:
            pass
            
    if channel:
        await channel.send(content)


async def reply_message(event: discord.Message, params: dict):
    """
    Executes: ACTION: REPLY_MESSAGE content="Hello"
    """
    content = params.get("content")
    if content:
        await event.channel.send(content, reference=event)


async def ban_user(event: discord.Message, params: dict):
    """
    Executes: ACTION: BAN_USER
    """
    # Safety: Don't ban the bot owner or admins (Basic protection)
    if event.author.guild_permissions.administrator:
        return 

    try:
        await event.author.ban(reason="GuildFlow Automated Ban")
    except discord.Forbidden:
        print(f"Error: Bot lacks permission to ban {event.author}")


async def timeout_user(event: discord.Message, params: dict):
    """
    Executes: ACTION: TIMEOUT_USER duration="1h"
    """
    duration_str = params.get("duration", "5m")
    duration_obj = parse_duration(duration_str)

    try:
        await event.author.timeout(duration_obj, reason="GuildFlow Automated Timeout")
    except discord.Forbidden:
        print(f"Error: Bot lacks permission to timeout {event.author}")


# --- THE COMMAND MAP ---
# This is what the Bot Engine will import tomorrow
COMMAND_MAP = {
    "SEND_MESSAGE": send_message,
    "REPLY_MESSAGE": reply_message,
    "BAN_USER": ban_user,
    "TIMEOUT_USER": timeout_user
}

```
