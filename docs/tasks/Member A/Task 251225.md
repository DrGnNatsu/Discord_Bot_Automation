### Part A: The Real Compiler (`src/compiler/compiler.py`)

*Replace the entire Mock file with this logic.*

```python
from src.antlr_build.GuildFlowParser import GuildFlowParser
from src.antlr_build.GuildFlowVisitor import GuildFlowVisitor

class GuildFlowCompiler(GuildFlowVisitor):
    
    def visitProg(self, ctx: GuildFlowParser.ProgContext):
        """Entry Point: Visits all definitions in the file."""
        return [self.visit(child) for child in ctx.definition()]

    def visitWorkflow_def(self, ctx: GuildFlowParser.Workflow_defContext):
        """
        Input: WORKFLOW anti_spam ON message { ... }
        Output: Dict with name, trigger, and list of actions.
        """
        return {
            "type": "WORKFLOW",
            "name": ctx.ID().getText(),
            "trigger": ctx.event_type().getText(),
            "filter": self.visit(ctx.condition()) if ctx.condition() else None,
            # RECURSION: Visit every statement inside the brackets
            "actions": [self.visit(stmt) for stmt in ctx.statement()]
        }

    def visitAction_statement(self, ctx: GuildFlowParser.Action_statementContext):
        """Pass-through to the command handler"""
        return self.visit(ctx.action_command())

    def visitAction_command(self, ctx: GuildFlowParser.Action_commandContext):
        """
        Input: ACTION: SEND_MESSAGE channel=general content="Hi"
        """
        command = ctx.start.text # 'SEND_MESSAGE', 'BAN_USER', etc.
        params = {}

        # Helper to strip quotes
        def clean(token): 
            return token.getText().strip('"') if token else None

        # Extract based on command type
        if command == "SEND_MESSAGE":
            if ctx.ID(): params["channel"] = ctx.ID().getText()
            if ctx.STRING(): params["content"] = clean(ctx.STRING())
        
        elif command == "TIMEOUT_USER":
            if ctx.STRING(): params["duration"] = clean(ctx.STRING())

        # (Add other commands like REPLY_MESSAGE here if needed)

        return {
            "type": "ACTION",
            "command": command,
            "params": params
        }

    def visitIf_statement(self, ctx: GuildFlowParser.If_statementContext):
        """
        Input: IF condition { stmt... } ELSE { stmt... }
        Output: Structure with 'then' and 'else' lists.
        """
        # Note: In the grammar, ctx.statement() returns a list of ALL statements.
        # We need to split them into THEN block and ELSE block.
        # This implementation assumes the standard structure where blocks are grouped.
        
        # Simple approach: We visit all children statements. 
        # For a robust implementation, we'd need named labels in .g4, 
        # but for now, we capture the structure:
        
        condition = self.visit(ctx.condition())
        
        # Logic: If specific grammar labels aren't used, obtaining distinct 
        # then/else blocks is tricky. We will return a simplified object 
        # assuming the Dispatcher handles the logic flow dynamically.
        
        # For this stage (MVP), we simplify to:
        return {
            "type": "LOGIC_IF",
            "condition": condition,
            # We recursively visit all statements found inside this context
            "body": [self.visit(stmt) for stmt in ctx.statement()]
        }

    def visitCondition(self, ctx: GuildFlowParser.ConditionContext):
        """Input: user.strikes >= 3"""
        return {
            "left": ctx.expr(0).getText(),
            "operator": ctx.comparator().getText(),
            "right": ctx.expr(1).getText()
        }

```
